# No classes, no abstraction

def render_graphic(graphic):
    # Manually check the "type" or structure
    if graphic['type'] == 'shape':
        print(f"Rendering Shape: {graphic['name']}")
        
    elif graphic['type'] == 'group':
        print("Starting Group Render...")
        for child in graphic['children']:
            # Recursion is still used, but the logic is centralized
            render_graphic(child) 
        print("End of Group Render.")

# Data is separated from logic
circle = {'type': 'shape', 'name': 'Circle'}
square = {'type': 'shape', 'name': 'Square'}

group = {
    'type': 'group',
    'children': [circle, square]
}

render_graphic(group)

In the procedural approach you would update the render_graphic() function for any new type.
shape_composite.py 

from abc import ABC, abstractmethod

# Component: Common interface for all objects
class Graphic(ABC):
    @abstractmethod
    def render(self):
        pass

# Leaf: Individual objects
class Shape(Graphic):
    def __init__(self, name):
        self.name = name

    def render(self):
        print(f"Rendering Shape: {self.name}")

# Composite: Container that can hold Leaves or other Composites
class GraphicGroup(Graphic):
    def __init__(self):
        self.children = []

    def add(self, graphic: Graphic):
        self.children.append(graphic)

    def render(self):
        print("Starting Group Render...")
        for child in self.children:
            child.render()
        print("End of Group Render.")

# Usage
circle = Shape("Circle")
square = Shape("Square")

group = GraphicGroup()
group.add(circle)
group.add(square)

# We treat the group exactly like a single shape
group.render()
How do we handle a single Circle differently than a group with 100 shapes?
No difference! The interface ensures the calling code doesn't have to care about the number or types of shapes.
