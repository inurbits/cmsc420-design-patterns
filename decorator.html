coffee_procedural.py 

# No classes, no abstraction

def get_cost(coffee):
    total = coffee['base_cost']
    if coffee.get('has_milk'):
        total += 2
    if coffee.get('has_sugar'):
        total += 1
    return total

# Data structure
my_coffee = {
    'base_cost': 5,
    'has_milk': True,
    'has_sugar': True
}

print(f"Total cost: ${get_cost(my_coffee)}")
How do I add a new coffee topping?
In the procedural approach you would update the get_cost() function and the my_coffee object for new toppings.
coffee_decorator.py 

from abc import ABC, abstractmethod

# Component interface
class Coffee(ABC):
    @abstractmethod
    def get_cost(self):
        pass

# Concrete Component
class SimpleCoffee(Coffee):
    def get_cost(self):
        return 5

# Base Decorator
class CoffeeDecorator(Coffee):
    def __init__(self, coffee: Coffee):
        self._coffee = coffee

    def get_cost(self):
        return self._coffee.get_cost()

# Concrete Decorators
class Milk(CoffeeDecorator):
    def get_cost(self):
        return self._coffee.get_cost() + 2

class Sugar(CoffeeDecorator):
    def get_cost(self):
        return self._coffee.get_cost() + 1

# Usage
my_coffee = SimpleCoffee()
my_coffee = Milk(my_coffee)
my_coffee = Sugar(my_coffee)

print(f"Total cost: ${my_coffee.get_cost()}") # Total: $8
How do we add a new topping?
Add a new concrete decorator class with our cost logic.
