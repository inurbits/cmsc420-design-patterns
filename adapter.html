Adapter Pattern

payment_procedural.py 
# No classes, no abstraction

def process_paypal_payment(paypal_email, amount_cents):
    print(f"PayPal: charging {paypal_email} ${amount_cents / 100:.2f}")

def process_stripe_payment(stripe_token, amount_cents):
    print(f"Stripe: charging token {stripe_token} ${amount_cents / 100:.2f}")


def checkout(cart_total, method, user_info):
	if method == "paypal":
		# Must know PayPal's exact function name and args
		process_paypal_payment(user_info["email"], cart_total * 100)
	elif method == "stripe":
		# Must know Stripe's exact function name and args
		process_stripe_payment(user_info["token"], cart_total * 100)
	else:
		raise ValueError(f"Unknown method: {method}")
	print("Order confirmed!")


# Usage
checkout(49.99, "paypal", {"email": "user@example.com"})
checkout(49.99, "stripe", {"token": "tok_abc123"})
How do I add a new provider?
In the procedural approach you'd add a new function like process_venmo_payment() and then edit checkout() to add another elif branch that calls it. Every new provider forces you to touch code that was already working!
payment_adapter.py 
from abc import ABC, abstractmethod


# ── Target interface ──────────────────
class PaymentProcessor(ABC):
    @abstractmethod
    def charge(self, amount: float) -> None:
        pass


# ── Third-party systems we can't change ──────────────────────
class PayPalAPI:
    def send_payment(self, email: str, cents: int):
        print(f"PayPal: charging {email} ${cents / 100:.2f}")

class StripeAPI:
    def create_charge(self, token: str, cents: int):
        print(f"Stripe: charging token {token} ${cents / 100:.2f}")

class VenmoAPI:
    def send_money(self, email: str, cents: int):
        print(f"Venmo: charging {email} ${cents / 100:.2f}")


# ── Adapters: translate .charge() to the provider's API ─────────────
class PayPalAdapter(PaymentProcessor):
    def __init__(self, email: str):
        self._api, self._email = PayPalAPI(), email

    def charge(self, amount: float) -> None:
        self._api.send_payment(self._email, int(amount * 100))


class StripeAdapter(PaymentProcessor):
    def __init__(self, token: str):
        self._api, self._token = StripeAPI(), token

    def charge(self, amount: float) -> None:
        self._api.create_charge(self._token, int(amount * 100))


class VenmoAdapter(PaymentProcessor):
    def __init__(self, email: str):
        self._api, self._email = VenmoAPI(), email

    def charge(self, amount: float) -> None:
        self._api.send_money(self._email, int(amount * 100))


# ── Checkout never changes, regardless of provider ───────────
def checkout(cart_total: float, processor: PaymentProcessor) -> None:
    processor.charge(cart_total)
    print("Order confirmed!")


# Usage — identical call for every provider
checkout(49.99, PayPalAdapter("user@example.com"))
checkout(49.99, StripeAdapter("tok_abc123"))
checkout(49.99, CryptoAdapter("1A2b3C4d..."))
How do I add a new provider?
Create a new class that extends PaymentProcessor and implements charge(). Inside it, instantiate the new provider's API and translate the call. Then pass an instance of your new adapter into checkout().
No existing code is modified.
